// ==UserScript==
// @name         Chess.com Bridge Bot
// @namespace    chess-bridge
// @version      1.5
// @description  Routes chess moves through a local Maia/Stockfish bridge, simulates realistic mouse drags, plays faster in the opening, has a draggable panel, and auto-queues the next game.
// @match        https://www.chess.com/play/*
// @match        https://www.chess.com/game/*
// @match        https://www.chess.com/daily
// @connect      127.0.0.1
// @connect      localhost
// @grant        GM_getValue
// @grant        GM_setValue
// @grant        GM_xmlhttpRequest
// @run-at       document-idle
// ==/UserScript==
(function () {
    'use strict';

    // ── Config ─────────────────────────────────────────────────────────────
    const CONFIG = {
        BOARD_SEL:  'chess-board, wc-chess-board',
        LOOP_MS:    200,
        BRIDGE_URL: 'http://127.0.0.1:8765/move',
        HEALTH_URL: 'http://127.0.0.1:8765/health',
    };

    // ── Persisted settings ─────────────────────────────────────────────────
    const settings = {
        autoRun:      GM_getValue('bot_autoRun',      true),
        autoMove:     GM_getValue('bot_autoMove',     true),
        minDelay:     GM_getValue('bot_minDelay',     0),
        maxDelay:     GM_getValue('bot_maxDelay',     0),
        forceProg:    GM_getValue('bot_forceProg',    false),
        openingBoost: GM_getValue('bot_openingBoost', true),
        openingMoves: GM_getValue('bot_openingMoves', 15),
        minimized:    GM_getValue('bot_minimized',    false),
        panelPos:     GM_getValue('bot_panelPos',     null),
        autoQueue:    GM_getValue('bot_autoQueue',    false),
    };
    const save = (k, v) => { settings[k] = v; GM_setValue(`bot_${k}`, v); };

    // ── State ──────────────────────────────────────────────────────────────
    const state = {
        board: null,
        isThinking: false,
        lastSanitizedBoardFEN: '',
        currentBestMove: null,
        moveTargetTime: 0,
        lastLiveResult: 'Initializing…',
        bridgeHealthy: null,
        lastEngine: null,
        lastOpponent: null,
        queueObserver: null,
        queueClicked: false,
    };

    // ── FEN helpers ────────────────────────────────────────────────────────
    function getRawBoardFEN() {
        if (!state.board?.game) return null;
        try {
            if (typeof state.board.game.getFEN === 'function') return state.board.game.getFEN();
            if (typeof state.board.game.fen === 'string') return state.board.game.fen;
            if (state.board.game.getPosition) return state.board.game.getPosition();
        } catch (e) {}
        return null;
    }

    function sanitizeFEN(rawFEN) {
        if (!rawFEN) return '';
        let parts = rawFEN.replace(/\s+/g, ' ').trim().split(' ');
        if (parts.length < 6) {
            const def = ['w', '-', '-', '0', '1'];
            for (let i = parts.length; i < 6; i++) parts.push(def[i - 1]);
        }
        if (parts[3] && parts[3] !== '-') parts[3] = parts[3].toLowerCase();
        return parts.join(' ');
    }

    // ═══ OPENING SPEED-UP ══════════════════════════════════════════════════
    function openingMultiplier(fen) {
        if (!settings.openingBoost) return 1.0;
        const N = settings.openingMoves;
        if (N <= 1) return 1.0;
        let fullmove = 1;
        try {
            const parts = fen.split(' ');
            fullmove = parseInt(parts[5], 10) || 1;
        } catch (e) {
            return 1.0;
        }
        if (fullmove >= N) return 1.0;
        const t = (fullmove - 1) / (N - 1);
        return 0.10 + t * 0.90;
    }

    function fenMoveNumber(fen) {
        try { return parseInt(fen.split(' ')[5], 10) || 1; }
        catch (e) { return 1; }
    }
    // ═══ END OPENING SPEED-UP ══════════════════════════════════════════════

    // ── Opponent detection ─────────────────────────────────────────────────
    function getOpponentUsername() {
        const me = (document.body.dataset.user || '').toLowerCase();
        const cards = document.querySelectorAll(
            '.player-component .user-username, ' +
            '[class*="player"] [class*="username"], ' +
            '.user-tagline-username, ' +
            '.player-user-info .username'
        );
        for (const el of cards) {
            const n = el.textContent.trim().toLowerCase();
            if (n && n !== me) return n;
        }
        const m = location.pathname.match(/\/play\/friend\/([^/?#]+)/);
        if (m && m[1].toLowerCase() !== me) return m[1].toLowerCase();
        return null;
    }

    // ── Bridge client ──────────────────────────────────────────────────────
    function requestBridgeMove(fen, opponent, cb) {
        GM_xmlhttpRequest({
            method: 'POST',
            url: CONFIG.BRIDGE_URL,
            headers: { 'Content-Type': 'application/json' },
            data: JSON.stringify({ fen, opponent: opponent || '' }),
            timeout: 15000,
            onload: (res) => {
                try { cb(JSON.parse(res.responseText)); }
                catch (e) { cb(null); }
            },
            onerror:   () => cb(null),
            ontimeout: () => cb(null),
        });
    }

    function checkBridgeHealth() {
        GM_xmlhttpRequest({
            method: 'GET',
            url: CONFIG.HEALTH_URL,
            timeout: 3000,
            onload: (res) => {
                try {
                    const d = JSON.parse(res.responseText);
                    state.bridgeHealthy = !!(d.maia_loaded && d.stockfish_loaded);
                    state.lastLiveResult = state.bridgeHealthy
                        ? 'Bridge healthy — Maia + Stockfish loaded'
                        : 'Bridge up but an engine failed to load';
                } catch {
                    state.bridgeHealthy = false;
                    state.lastLiveResult = 'Bridge responded but payload is invalid';
                }
                updateUI();
            },
            onerror: () => {
                state.bridgeHealthy = false;
                state.lastLiveResult = 'Bridge not reachable on 127.0.0.1:8765';
                updateUI();
            },
            ontimeout: () => {
                state.bridgeHealthy = false;
                state.lastLiveResult = 'Bridge health check timed out';
                updateUI();
            },
        });
    }

    // ── Analyze ────────────────────────────────────────────────────────────
    function analyze() {
        if (state.isThinking) return;
        const fen = sanitizeFEN(getRawBoardFEN());
        if (!fen) return;
        state.lastSanitizedBoardFEN = fen;
        state.isThinking = true;

        const minMs = settings.minDelay * 1000;
        const maxMs = settings.maxDelay * 1000;
        const baseDelay = Math.random() * (maxMs - minMs) + minMs;
        const mult = openingMultiplier(fen);
        const finalDelay = baseDelay * mult;
        state.moveTargetTime = performance.now() + finalDelay;

        const opponent = getOpponentUsername();
        state.lastOpponent = opponent;

        const moveNum = fenMoveNumber(fen);
        const speedNote = mult < 0.95
            ? ` · m${moveNum} @ ${Math.round(mult * 100)}% delay`
            : '';
        state.lastLiveResult = `Asking bridge${speedNote}…`;
        updateUI();

        requestBridgeMove(fen, opponent, (data) => {
            state.isThinking = false;

            if (data && data.uci) {
                state.bridgeHealthy = true;
                state.lastEngine = data.engine;
                state.currentBestMove = data.uci;

                const icon  = data.engine === 'maia' ? '👤' : '🎯';
                const label = data.engine === 'maia' ? 'Maia' : 'Stockfish';
                state.lastLiveResult = `${icon} ${label} vs ${opponent || '?'} → ${data.uci}${speedNote}`;

                if (settings.autoMove) triggerAutoMove();
            } else {
                state.bridgeHealthy = false;
                state.lastLiveResult = 'Bridge returned no move — is bridge.py running?';
            }
            updateUI();
        });
    }

    // ── Auto-move gate ─────────────────────────────────────────────────────
    function triggerAutoMove() {
        if (!state.currentBestMove || !state.board?.game) return;
        const turn = state.board.game.getTurn();
        const playingAs = state.board.game.getPlayingAs();
        if (turn !== playingAs) return;
        const wait = Math.max(0, state.moveTargetTime - performance.now());
        setTimeout(() => playMove(state.currentBestMove), wait);
    }

    // ═══ AUTO-QUEUE ════════════════════════════════════════════════════════
    // Chess.com has no public API for starting a game, so we watch the DOM
    // for the game-over modal and click its "New Game" / "Rematch" button.
    // Two guards make this safe:
    //   1. Only fires when a game-over container is actually present, so
    //      browsing the Play menu won't trigger anything.
    //   2. Only clicks buttons whose text is exactly "New Game" or "Rematch"
    //      and which live inside a modal/game-over/dialog container.

    function startAutoQueue() {
        if (state.queueObserver) return;
        state.queueObserver = new MutationObserver(handleQueueMutation);
        state.queueObserver.observe(document.body, { childList: true, subtree: true });
        handleQueueMutation(); // run once in case a modal is already up
    }

    function stopAutoQueue() {
        if (state.queueObserver) {
            state.queueObserver.disconnect();
            state.queueObserver = null;
        }
        state.queueClicked = false;
    }

function handleQueueMutation() {
    if (!settings.autoQueue) return;

    const anyGameOver = document.querySelector(
        '.game-over-modal-content, [class*="game-over"]'
    );
    if (!anyGameOver) {
        state.queueClicked = false;
        return;
    }
    if (state.queueClicked) return;

    const buttons = document.querySelectorAll('button, [role="button"]');
    let newGameBtn = null;

    for (const b of buttons) {
        const text = (b.textContent || '').trim().toLowerCase();

        // Explicitly skip rematch — sending rematch requests is not
        // what "auto-queue" means; it pins you to the same opponent.
        if (text.includes('rematch')) continue;

        // Accept "New Game", "New 5 min game", "Play new game", etc.
        // Anything that mentions both "new" and "game" qualifies.
        const isNewGame = text.includes('new') && text.includes('min');
        if (!isNewGame) continue;

        const visible =
            b.offsetParent !== null &&
            b.getBoundingClientRect().width > 0;
        if (!visible) continue;

        const inModal = b.closest(
            '[class*="modal"], [class*="game-over"], [class*="game-result"], [class*="dialog"]'
        );
        if (!inModal) continue;

        newGameBtn = b;
        break;
    }

    if (!newGameBtn) {
        // Nothing matched. Don't spam the status, but do log once so
        // you can diagnose with the console if it ever stops working.
        if (!state.queueLoggedNoButton) {
            state.queueLoggedNoButton = true;
            const visibleTexts = [...document.querySelectorAll('button, [role="button"]')]
                .filter(b => b.offsetParent !== null)
                .map(b => (b.textContent || '').trim())
                .filter(Boolean);
            console.log('[bridge] auto-queue: no New Game button found. Visible:', visibleTexts);
        }
        return;
    }
    state.queueLoggedNoButton = false;

    state.queueClicked = true;
    const clickedText = newGameBtn.textContent.trim();
    const delayMs = 800 + Math.random() * 2200;
    state.lastLiveResult = `🔄 Auto-queue "${clickedText}" in ${(delayMs / 1000).toFixed(1)}s…`;
    updateUI();

    setTimeout(() => {
        try {
            newGameBtn.click();
            state.lastLiveResult = `🔄 Clicked "${clickedText}" — queuing next game`;
        } catch (e) {
            state.lastLiveResult = `⚠️ Auto-queue click failed: ${e.message}`;
        }
        updateUI();
    }, delayMs);
}
    // ═══ END AUTO-QUEUE ════════════════════════════════════════════════════

    // ═══ BOARD GEOMETRY + MOUSE DRAG ═══════════════════════════════════════

    const sleep = (ms) => new Promise(r => setTimeout(r, ms));

    function locateBoard() {
        const wc = document.querySelector('wc-chess-board');
        if (wc) {
            let inner = null;
            try {
                inner = wc.shadowRoot?.querySelector('cg-board')
                     || wc.shadowRoot?.querySelector('chess-board')
                     || wc.shadowRoot?.querySelector('.board');
            } catch (e) {}
            if (inner) return { el: inner, host: wc };
            return { el: wc, host: null };
        }
        const cg = document.querySelector('cg-board');
        if (cg) return { el: cg, host: null };
        const cb = document.querySelector('chess-board');
        if (cb) {
            const inner = cb.querySelector('cg-board') || cb.querySelector('.board');
            return { el: inner || cb, host: null };
        }
        return null;
    }

    function boardIsFlipped(boardInfo) {
        if (boardInfo.host?.classList?.contains('flipped')) return true;
        if (boardInfo.el.classList?.contains('flipped')) return true;
        try {
            if (boardInfo.el.closest?.('.flipped')) return true;
        } catch (e) {}
        try {
            if (state.board?.game?.getPlayingAs) {
                return state.board.game.getPlayingAs() === 2;
            }
        } catch (e) {}
        return false;
    }

    function squareToPixel(boardInfo, square, flipped) {
        const r = boardInfo.el.getBoundingClientRect();
        if (r.width < 50 || r.height < 50) return null;
        const sqW = r.width / 8, sqH = r.height / 8;
        const fileIdx = square.charCodeAt(0) - 97;
        const rankIdx = parseInt(square[1], 10) - 1;
        const col = flipped ? 7 - fileIdx : fileIdx;
        const row = flipped ? rankIdx     : 7 - rankIdx;
        return {
            x: r.left + col * sqW + sqW / 2,
            y: r.top  + row * sqH + sqH / 2,
        };
    }

    function fire(boardEl, type, x, y, buttons) {
        const evt = new PointerEvent(type, {
            bubbles: true, cancelable: true, composed: true,
            clientX: x, clientY: y,
            screenX: x + (window.screenX || 0),
            screenY: y + (window.screenY || 0),
            pointerId: 1, pointerType: 'mouse', isPrimary: true,
            button: 0, buttons,
        });
        boardEl.dispatchEvent(evt);
    }

    function generateTrajectory(a, b) {
        const dx = b.x - a.x, dy = b.y - a.y;
        const dist = Math.hypot(dx, dy) || 1;
        const n = Math.max(5, Math.min(22, Math.round(dist / 14)));

        const px = -dy / dist, py = dx / dist;
        const curveMag = dist * (0.05 + Math.random() * 0.18) * (Math.random() < 0.5 ? -1 : 1);
        const c1 = { x: a.x + dx * 0.25 + px * curveMag * 0.8,
                     y: a.y + dy * 0.25 + py * curveMag * 0.8 };
        const c2 = { x: a.x + dx * 0.75 + px * curveMag * 0.2,
                     y: a.y + dy * 0.75 + py * curveMag * 0.2 };

        const pts = [];
        for (let i = 0; i <= n; i++) {
            let t = i / n;
            t = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
            const u = 1 - t;
            const x = u*u*u * a.x + 3*u*u*t * c1.x + 3*u*t*t * c2.x + t*t*t * b.x;
            const y = u*u*u * a.y + 3*u*u*t * c1.y + 3*u*t*t * c2.y + t*t*t * b.y;
            pts.push({ x: x + (Math.random() - 0.5) * 1.4,
                       y: y + (Math.random() - 0.5) * 1.4 });
        }
        return pts;
    }

    async function simulateDrag(fromSq, toSq) {
        const boardInfo = locateBoard();
        if (!boardInfo) return { ok: false, reason: 'board element not found' };

        const flipped = boardIsFlipped(boardInfo);
        const from = squareToPixel(boardInfo, fromSq, flipped);
        const to   = squareToPixel(boardInfo, toSq,   flipped);
        if (!from) return { ok: false, reason: 'source square off-board' };
        if (!to)   return { ok: false, reason: 'target square off-board' };

        const boardEl = boardInfo.el;

        fire(boardEl, 'pointerdown', from.x, from.y, 1);
        await sleep(30 + Math.random() * 50);

        const path = generateTrajectory(from, to);
        for (let i = 1; i < path.length; i++) {
            await sleep(6 + Math.random() * 12);
            fire(boardEl, 'pointermove', path[i].x, path[i].y, 1);
        }

        await sleep(30 + Math.random() * 40);
        fire(boardEl, 'pointerup', to.x, to.y, 0);

        return { ok: true, flipped };
    }

    async function waitForFenChange(oldFen, timeoutMs = 1500) {
        const start = performance.now();
        const oldBoard = oldFen.split(' ')[0];
        while (performance.now() - start < timeoutMs) {
            const current = sanitizeFEN(getRawBoardFEN());
            if (current && current.split(' ')[0] !== oldBoard) return true;
            await sleep(50);
        }
        return false;
    }

    async function clickPromotionPiece(piece) {
        const start = performance.now();
        const selectors = [
            `.promotion-window [data-piece="${piece}"]`,
            `.promotion-window .${piece}`,
            `.promotion-window [class*="${piece}"]`,
            `[class*="promotion"] [data-piece="${piece}"]`,
            `[class*="promotion"] .${piece}`,
            `[class*="promotion"] [class*="${piece}"]`,
        ];
        while (performance.now() - start < 1500) {
            for (const sel of selectors) {
                const el = document.querySelector(sel);
                if (el) { el.click(); return true; }
            }
            await sleep(60);
        }
        return false;
    }

    function fallbackMove(move) {
        const from = move.substring(0, 2), to = move.substring(2, 4);
        const promo = move.length > 4 ? move.substring(4, 5) : 'q';
        try {
            for (const m of state.board.game.getLegalMoves()) {
                if (m.from === from && m.to === to) {
                    state.board.game.move({ ...m, promotion: promo, animate: true, userGenerated: true });
                    state.lastLiveResult = `Played ${from}→${to} (programmatic fallback)`;
                    updateUI();
                    return true;
                }
            }
        } catch (e) {
            state.lastLiveResult = `Fallback error: ${e.message}`;
            updateUI();
        }
        return false;
    }

    async function playMove(move) {
        try {
            if (!state.board?.game) return;
            const from = move.substring(0, 2);
            const to   = move.substring(2, 4);
            const promo = move.length > 4 ? move.substring(4, 5) : 'q';

            const fenBefore = sanitizeFEN(getRawBoardFEN());
            if (!fenBefore ||
                fenBefore.split(' ')[0] !== state.lastSanitizedBoardFEN.split(' ')[0]) {
                return;
            }

            if (settings.forceProg) {
                fallbackMove(move);
                return;
            }

            state.lastLiveResult = `🐭 Dragging ${from}→${to}…`;
            updateUI();

            let dragged = false;
            try {
                const result = await simulateDrag(from, to);
                dragged = result.ok;
                if (!dragged) {
                    state.lastLiveResult = `⚠️ Drag: ${result.reason}`;
                    updateUI();
                }
            } catch (e) {
                console.error('[bridge] drag threw:', e);
            }

            if (dragged) {
                const changed = await waitForFenChange(fenBefore, 1500);
                if (changed) {
                    const isPromotion = (from[1] === '7' && to[1] === '8') ||
                                        (from[1] === '2' && to[1] === '1');
                    if (isPromotion) await clickPromotionPiece(promo);
                    state.lastLiveResult = `Played ${from}→${to} (drag)`;
                    updateUI();
                    return;
                }
                state.lastLiveResult = `Drag didn't register — using fallback`;
                updateUI();
            }

            if (!fallbackMove(move)) {
                state.lastSanitizedBoardFEN = '';
            }
        } catch (e) {
            console.error('[bridge] playMove error:', e);
            state.lastLiveResult = `Error: ${e.message}`;
            state.lastSanitizedBoardFEN = '';
            updateUI();
        }
    }

    // ── Diagnose ───────────────────────────────────────────────────────────
    function diagnoseBoard() {
        const info = locateBoard();
        if (!info) {
            state.lastLiveResult = 'Board not found — inspect DOM manually';
            updateUI();
            return;
        }
        const r = info.el.getBoundingClientRect();
        const flipped = boardIsFlipped(info);
        const e2 = squareToPixel(info, 'e2', flipped);
        const e4 = squareToPixel(info, 'e4', flipped);
        const tag = info.el.tagName.toLowerCase();
        const shadow = info.host ? '(shadow)' : '';

        // Also list visible button texts — useful for tuning auto-queue
        const btnTexts = [...document.querySelectorAll('button, [role="button"]')]
            .filter(b => b.offsetParent !== null)
            .map(b => (b.textContent || '').trim())
            .filter(Boolean)
            .slice(0, 12);
        const btnList = btnTexts.length ? ` btn=[${btnTexts.join(' | ')}]` : ' btn=[]';

        state.lastLiveResult =
            `board=<${tag}>${shadow} ` +
            `rect=${Math.round(r.left)},${Math.round(r.top)} ` +
            `${Math.round(r.width)}x${Math.round(r.height)} ` +
            `flipped=${flipped} ` +
            `e2=${e2 ? `${Math.round(e2.x)},${Math.round(e2.y)}` : '—'} ` +
            `e4=${e4 ? `${Math.round(e4.x)},${Math.round(e4.y)}` : '—'}` +
            btnList;
        updateUI();
    }

    // ═══ END BOARD GEOMETRY ════════════════════════════════════════════════

    // ═══ DRAGGABLE + MINIMIZABLE PANEL ═════════════════════════════════════
    function clampPanelToViewport(panel) {
        if (!panel) return;
        const r = panel.getBoundingClientRect();
        const maxX = window.innerWidth  - r.width;
        const maxY = window.innerHeight - r.height;
        const left = Math.max(0, Math.min(maxX, r.left));
        const top  = Math.max(0, Math.min(maxY, r.top));
        panel.style.left = left + 'px';
        panel.style.top  = top  + 'px';
        panel.style.right  = 'auto';
        panel.style.bottom = 'auto';
    }

    function applySavedPanelState(panel, header) {
        if (settings.panelPos) {
            panel.style.left  = settings.panelPos.left + 'px';
            panel.style.top   = settings.panelPos.top  + 'px';
            panel.style.right = 'auto';
            panel.style.bottom = 'auto';
        }
        if (settings.minimized) {
            panel.classList.add('minimized');
            header.querySelector('#bridgeMin').textContent = '+';
        }
    }

    function attachDragHandlers(panel, header) {
        let drag = null;

        header.addEventListener('pointerdown', (e) => {
            if (e.target.closest('.hdr-btn')) return;
            e.preventDefault();
            const r = panel.getBoundingClientRect();
            drag = {
                startX: e.clientX,
                startY: e.clientY,
                startLeft: r.left,
                startTop: r.top,
            };
            header.setPointerCapture(e.pointerId);
            header.style.cursor = 'grabbing';
        });

        header.addEventListener('pointermove', (e) => {
            if (!drag) return;
            const nextLeft = drag.startLeft + (e.clientX - drag.startX);
            const nextTop  = drag.startTop  + (e.clientY - drag.startY);
            panel.style.left  = nextLeft + 'px';
            panel.style.top   = nextTop  + 'px';
            panel.style.right = 'auto';
            panel.style.bottom = 'auto';
            clampPanelToViewport(panel);
        });

        const endDrag = (e) => {
            if (!drag) return;
            drag = null;
            header.style.cursor = '';
            try { header.releasePointerCapture(e.pointerId); } catch (_) {}
            save('panelPos', { left: panel.offsetLeft, top: panel.offsetTop });
        };
        header.addEventListener('pointerup', endDrag);
        header.addEventListener('pointercancel', endDrag);

        window.addEventListener('resize', () => {
            if (panel.isConnected) clampPanelToViewport(panel);
        });
    }

    function toggleMinimize(panel) {
        const nowMin = panel.classList.toggle('minimized');
        save('minimized', nowMin);
        const btn = panel.querySelector('#bridgeMin');
        if (btn) btn.textContent = nowMin ? '+' : '−';
        if (!nowMin) clampPanelToViewport(panel);
    }
    // ═══ END DRAGGABLE + MINIMIZABLE PANEL ═════════════════════════════════

    // ── UI ─────────────────────────────────────────────────────────────────
    function createUI() {
        if (document.getElementById('bridgePanel')) return;

        const style = document.createElement('style');
        style.textContent = `
            #bridgePanel {

    function updateUI() {
        const statusEl = document.getElementById('bridgeStatus');
        if (!statusEl) return;

        const cls = state.bridgeHealthy === true  ? 'green'
                  : state.bridgeHealthy === false ? 'red'
                  :                                 'gray';

        const hdrDot = document.getElementById('bridgeHdrDot');
        if (hdrDot) hdrDot.className = `hdr-dot ${cls}`;

        const engineTag = state.lastEngine
            ? (state.lastEngine === 'maia' ? '👤 Maia' : '🎯 Stockfish')
            : '—';

        statusEl.innerHTML = `
            <div><span class="hdr-dot ${cls}" style="margin-right:5px; vertical-align:middle;"></span>${escapeHtml(state.lastLiveResult)}</div>
            <div style="color:#888; margin-top:4px; font-size:10px;">
                engine: ${engineTag}${state.lastOpponent ? ` · vs ${escapeHtml(state.lastOpponent)}` : ''}
            </div>
        `;
    }

    function escapeHtml(s) {
        return String(s).replace(/[&<>"']/g, c =>
            ({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[c]));
    }

    // ── Main loop ──────────────────────────────────────────────────────────
    function mainLoop() {
        if (!document.getElementById('bridgePanel')) createUI();

        state.board = document.querySelector(CONFIG.BOARD_SEL);
        if (!state.board) {
            setTimeout(mainLoop, CONFIG.LOOP_MS);
            return;
        }

        if (state.board.game && settings.autoRun) {
            const raw = getRawBoardFEN();
            if (raw) {
                const clean = sanitizeFEN(raw);
                const isTurn = state.board.game.getTurn() === state.board.game.getPlayingAs();
                if (isTurn && clean !== state.lastSanitizedBoardFEN) analyze();
            }
        }

        setTimeout(mainLoop, CONFIG.LOOP_MS);
    }

    // ── Boot ───────────────────────────────────────────────────────────────
    mainLoop();
    setTimeout(checkBridgeHealth, 800);
})();
